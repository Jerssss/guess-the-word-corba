-- =======================
-- TABLE: players
-- =======================
CREATE TABLE players (
    player_id INT AUTO_INCREMENT PRIMARY KEY,
    full_name VARCHAR(100) NOT NULL,
    username VARCHAR(50) NOT NULL UNIQUE,
    password VARCHAR(50) NOT NULL,
    game_wins INT DEFAULT 0,
    is_logged_in TINYINT(1) DEFAULT 0  -- 0 = false, 1 = true
) ENGINE=InnoDB;

-- =======================
-- TABLE: games
-- =======================
CREATE TABLE games (
    game_id INT AUTO_INCREMENT PRIMARY KEY,
    total_rounds INT,
    game_winner INT,
    game_status ENUM('waiting', 'in_progress', 'completed', 'cancelled') DEFAULT 'waiting',
    start_time DATETIME DEFAULT CURRENT_TIMESTAMP,
    end_time DATETIME,
    FOREIGN KEY (game_winner) REFERENCES players(player_id)
        ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB;

-- =======================
-- TABLE: game_players (Join table)
-- =======================
CREATE TABLE game_players (
    game_id INT,
    player_id INT,
    PRIMARY KEY (game_id, player_id),
    FOREIGN KEY (game_id) REFERENCES games(game_id)
        ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (player_id) REFERENCES players(player_id)
        ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB;

-- =======================
-- TABLE: rounds
-- =======================
CREATE TABLE rounds (
    round_id INT AUTO_INCREMENT PRIMARY KEY,
    game_id INT,
    round_number INT NOT NULL,
    word VARCHAR(50) NOT NULL,
    word_length INT NOT NULL,
    round_winner INT,
    start_time DATETIME DEFAULT CURRENT_TIMESTAMP,
    end_time DATETIME,
    FOREIGN KEY (game_id) REFERENCES games(game_id)
        ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (round_winner) REFERENCES players(player_id)
        ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB;

-- =======================
-- TABLE: round_guesses
-- =======================
CREATE TABLE round_guesses (
    guess_id INT AUTO_INCREMENT PRIMARY KEY,
    round_id INT,
    player_id INT,
    guessed_letter CHAR(1),
    is_correct TINYINT(1) DEFAULT 0,  -- 0 = false, 1 = true
    guess_time DATETIME DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (round_id) REFERENCES rounds(round_id)
        ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (player_id) REFERENCES players(player_id)
        ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB;

-- =======================
-- TABLE: round_stats (optional – tracks incorrect attempts)
-- =======================
CREATE TABLE round_stats (
    round_id INT,
    player_id INT,
    incorrect_attempts INT DEFAULT 0,
    PRIMARY KEY (round_id, player_id),
    FOREIGN KEY (round_id) REFERENCES rounds(round_id)
        ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (player_id) REFERENCES players(player_id)
        ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB;

-- =======================
-- TABLE: settings
-- =======================
CREATE TABLE settings (
    setting_name VARCHAR(50) PRIMARY KEY,
    waiting_time INT DEFAULT 10,       -- seconds before game starts
    round_duration INT DEFAULT 30      -- seconds per round
) ENGINE=InnoDB;
